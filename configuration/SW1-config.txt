! ============================================================
! SW1 - Core/Distribution Switch Configuration
! Secure Enterprise Network
! ============================================================

hostname SW1

! -------------------------
! Basic Security
! -------------------------
enable secret Class@123

line console 0
 password Console@123
 login
 exit

! -------------------------
! VLAN Configuration
! -------------------------
vlan 10
 name HR
 exit

vlan 20
 name IT
 exit

vlan 30
 name SALES
 exit

vlan 40
 name SERVERS
 exit

vlan 99
 name MANAGEMENT
 exit

! -------------------------
! Management SVI
! -------------------------
interface vlan 99
 ip address 192.168.99.10 255.255.255.0
 no shutdown
 exit

ip default-gateway 192.168.99.1

! -------------------------
! SSH Configuration
! -------------------------
ip domain-name enterprise.local

username admin privilege 15 secret Cisco@123

crypto key generate rsa
1024

ip ssh version 2

access-list 10 permit 192.168.99.0 0.0.0.255

line vty 0 4
 login local
 transport input ssh
 access-class 10 in
 exit

! -------------------------
! Server Port
! -------------------------
interface fa0/10
 switchport mode access
 switchport access vlan 40
 exit

! -------------------------
! Trunk to SW2
! -------------------------
interface fa0/1
 switchport mode trunk
 exit

! -------------------------
! Trunk to SW3
! -------------------------
interface fa0/2
 switchport mode trunk
 exit

! -------------------------
! Trunk to Router
! -------------------------
interface gigabitEthernet 0/1
 switchport mode trunk
 exit

! -------------------------
! Disable Unused Ports
! -------------------------
interface range fa0/3 - 9
 shutdown
 exit

interface range fa0/11 - 24
 shutdown
 exit

! -------------------------
! DHCP Snooping
! -------------------------
ip dhcp snooping
ip dhcp snooping vlan 10,20,30

interface range fa0/1 - 2
 ip dhcp snooping trust
 exit

interface gigabitEthernet 0/1
 ip dhcp snooping trust
 exit

no ip dhcp snooping information option

! -------------------------
! Dynamic ARP Inspection
! -------------------------
ip arp inspection vlan 10,20,30

interface range fa0/1 - 2
 ip arp inspection trust
 exit

interface gigabitEthernet 0/1
 ip arp inspection trust
 exit

! -------------------------
! Syslog
! -------------------------
logging 192.168.40.10

service timestamps log datetime msec

! -------------------------
! End
! -------------------------
end
